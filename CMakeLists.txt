cmake_minimum_required(VERSION 3.25)
project(AsrWin LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AsrWin targets Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(READ "${CMAKE_SOURCE_DIR}/shipped-model.json" SHIPPED_MODEL_JSON)
string(JSON ASRWIN_MODEL_DIR GET "${SHIPPED_MODEL_JSON}" model_dir)
string(JSON ASRWIN_THREADS ERROR_VARIABLE THREADS_ERROR GET "${SHIPPED_MODEL_JSON}" threads)
if(THREADS_ERROR)
  set(ASRWIN_THREADS 6)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/shipped-model.json")
# Provider policy and the optional large (GPU-only) model are compiled in from the same file.
string(JSON ASRWIN_DEFAULT_PROVIDER ERROR_VARIABLE PROVIDER_ERROR GET "${SHIPPED_MODEL_JSON}" default_provider)
if(PROVIDER_ERROR)
  set(ASRWIN_DEFAULT_PROVIDER "cpu")
endif()
if(NOT ASRWIN_DEFAULT_PROVIDER MATCHES "^(cpu|directml)$")
  message(FATAL_ERROR "shipped-model.json default_provider must be cpu or directml")
endif()
string(JSON DIRECTML_VALIDATED_JSON ERROR_VARIABLE VALIDATED_ERROR GET "${SHIPPED_MODEL_JSON}" directml validated)
if(VALIDATED_ERROR OR NOT DIRECTML_VALIDATED_JSON)
  set(ASRWIN_DIRECTML_VALIDATED 0)
else()
  set(ASRWIN_DIRECTML_VALIDATED 1)
endif()
if(ASRWIN_DEFAULT_PROVIDER STREQUAL "directml" AND NOT ASRWIN_DIRECTML_VALIDATED)
  message(FATAL_ERROR "shipped-model.json: default_provider directml requires directml.validated true")
endif()
string(JSON ASRWIN_LARGE_MODEL_DIR ERROR_VARIABLE LARGE_ERROR GET "${SHIPPED_MODEL_JSON}" directml large_model model_dir)
if(LARGE_ERROR)
  set(ASRWIN_LARGE_MODEL_DIR "")
endif()
string(JSON ASRWIN_LARGE_MODEL_THREADS ERROR_VARIABLE LARGE_THREADS_ERROR GET "${SHIPPED_MODEL_JSON}" directml large_model threads)
if(LARGE_THREADS_ERROR)
  set(ASRWIN_LARGE_MODEL_THREADS ${ASRWIN_THREADS})
endif()
string(JSON ASRWIN_MIN_VRAM_BYTES ERROR_VARIABLE MIN_VRAM_ERROR GET "${SHIPPED_MODEL_JSON}" minimum_dedicated_video_memory_bytes)
if(MIN_VRAM_ERROR)
  set(ASRWIN_MIN_VRAM_BYTES 0)
endif()
string(JSON ASRWIN_LARGE_MODEL_MIN_VRAM_BYTES ERROR_VARIABLE LARGE_MIN_VRAM_ERROR GET "${SHIPPED_MODEL_JSON}" directml large_model minimum_dedicated_video_memory_bytes)
if(LARGE_MIN_VRAM_ERROR)
  set(ASRWIN_LARGE_MODEL_MIN_VRAM_BYTES 0)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
function(dep_source NAME OUT)
  file(READ "${CMAKE_SOURCE_DIR}/dependencies.lock.json" DEPENDENCY_LOCK)
  string(JSON REVISION GET "${DEPENDENCY_LOCK}" sources "${NAME}" revision)
  string(JSON REPOSITORY GET "${DEPENDENCY_LOCK}" sources "${NAME}" repo)
  get_filename_component(REPOSITORY_NAME "${REPOSITORY}" NAME)
  set(DIR "${CMAKE_SOURCE_DIR}/.deps/${NAME}/${REPOSITORY_NAME}-${REVISION}")
  if(NOT IS_DIRECTORY "${DIR}")
    message(FATAL_ERROR "Missing pinned ${NAME} at ${REVISION}; run scripts/bootstrap.py first.")
  endif()
  set(${OUT} "${DIR}" PARENT_SCOPE)
endfunction()
dep_source(kissfft KISS)
dep_source(libfvad FVAD)
dep_source(tokenizers-cpp TOK)
dep_source(json JSON)
set(ORT "${CMAKE_SOURCE_DIR}/.deps/onnxruntime")
set(DML "${CMAKE_SOURCE_DIR}/.deps/directml")
set(ASRWIN_REDIST_DIR "" CACHE PATH "Pinned Microsoft VC 14.51.36231 native x64 CRT redistributables")
if(NOT EXISTS "${ASRWIN_REDIST_DIR}/msvcp140.dll")
  message(FATAL_ERROR "Set ASRWIN_REDIST_DIR to the pinned x64 VC CRT directory or use scripts/build.ps1")
endif()
add_library(kissfft STATIC "${KISS}/kiss_fft.c" "${KISS}/kiss_fftr.c")
target_include_directories(kissfft PUBLIC "${KISS}")
file(GLOB_RECURSE FVAD_SOURCES CONFIGURE_DEPENDS "${FVAD}/src/*.c")
add_library(fvad STATIC ${FVAD_SOURCES})
target_include_directories(fvad PUBLIC "${FVAD}/include" PRIVATE "${FVAD}/src")
set(TOK_LIB "${CMAKE_BINARY_DIR}/tokenizer/release/tokenizers_c.lib")
find_program(CARGO cargo REQUIRED)
configure_file("${CMAKE_SOURCE_DIR}/tokenizer.Cargo.lock" "${TOK}/rust/Cargo.lock" COPYONLY)
add_custom_command(OUTPUT "${TOK_LIB}"
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_BINARY_DIR}/tokenizer" "RUSTFLAGS=-Ctarget-feature=-crt-static" "CARGO_HOME=${CMAKE_SOURCE_DIR}/.cache/cargo" ${CARGO} build --locked --release --manifest-path "${TOK}/rust/Cargo.toml"
  WORKING_DIRECTORY "${TOK}/rust" VERBATIM)
add_custom_target(tokenizer_rust DEPENDS "${TOK_LIB}")
add_library(tokenizers_cpp STATIC "${TOK}/src/huggingface_tokenizer.cc")
target_include_directories(tokenizers_cpp PUBLIC "${TOK}/include")
add_dependencies(tokenizers_cpp tokenizer_rust)
target_link_libraries(tokenizers_cpp PUBLIC "${TOK_LIB}" ntdll wsock32 ws2_32 bcrypt iphlpapi userenv psapi)
add_executable(AsrWin src/main.cpp src/common.cpp src/audio.cpp src/mel.cpp src/asr.cpp src/scoring.cpp src/diagnostics.cpp src/compute.cpp src/settings.cpp src/segmenter.cpp src/capture.cpp src/pipeline.cpp src/gui.cpp src/self_test.cpp)
target_link_options(AsrWin PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/src/AsrWin.manifest")
target_compile_definitions(AsrWin PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_definitions(AsrWin PRIVATE ASRWIN_MODEL_DIR="${ASRWIN_MODEL_DIR}")
target_compile_definitions(AsrWin PRIVATE ASRWIN_THREADS=${ASRWIN_THREADS})
target_compile_definitions(AsrWin PRIVATE ASRWIN_DEFAULT_PROVIDER="${ASRWIN_DEFAULT_PROVIDER}" ASRWIN_DIRECTML_VALIDATED=${ASRWIN_DIRECTML_VALIDATED})
target_compile_definitions(AsrWin PRIVATE ASRWIN_LARGE_MODEL_DIR="${ASRWIN_LARGE_MODEL_DIR}" ASRWIN_LARGE_MODEL_THREADS=${ASRWIN_LARGE_MODEL_THREADS})
target_compile_definitions(AsrWin PRIVATE ASRWIN_MIN_VRAM_BYTES=${ASRWIN_MIN_VRAM_BYTES}ull ASRWIN_LARGE_MODEL_MIN_VRAM_BYTES=${ASRWIN_LARGE_MODEL_MIN_VRAM_BYTES}ull)
target_compile_options(AsrWin PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_include_directories(AsrWin PRIVATE "${JSON}/single_include" "${ORT}/build/native/include" "${DML}/include")
target_link_libraries(AsrWin PRIVATE kissfft fvad tokenizers_cpp "${ORT}/runtimes/win-x64/native/onnxruntime.lib" ole32 mfplat mf mfuuid wmcodecdspuuid propsys avrt shlwapi comctl32 comdlg32 psapi bcrypt normaliz)
target_link_libraries(AsrWin PRIVATE dxgi shell32)
file(GLOB DML_DLL "${DML}/bin/x64-win/DirectML.dll")
if(NOT DML_DLL)
  message(FATAL_ERROR "DirectML.dll missing; run scripts/native_deps.py")
endif()

  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT}/runtimes/win-x64/native/onnxruntime.dll" "$<TARGET_FILE_DIR:AsrWin>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT}/runtimes/win-x64/native/onnxruntime_providers_shared.dll" "$<TARGET_FILE_DIR:AsrWin>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DML_DLL}" "$<TARGET_FILE_DIR:AsrWin>")

  add_custom_command(TARGET AsrWin POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CRT_DLL}" "$<TARGET_FILE_DIR:AsrWin>")
endforeach()
